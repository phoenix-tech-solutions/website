import React, { useRef, useState, useEffect } from 'react'

const HeroSection = () => {
  const [hovered, setHovered] = useState(false)
  const [scrollProgress, setScrollProgress] = useState(0)
  const [phoenixComplete, setPhoenixComplete] = useState(false)
  const [allowScroll, setAllowScroll] = useState(false)
  const canvasRef = useRef<HTMLCanvasElement | null>(null)
  const animationRef = useRef<number | null>(null)
  const sectionRef = useRef<HTMLDivElement | null>(null)
  const scrollLocked = useRef(false)
  const accumulatedScroll = useRef(0)
  const lastScrollY = useRef(0)
  
  type Particle = {
    x: number; y: number; targetX: number; targetY: number;
    phoenixX: number; phoenixY: number; size: number; opacity: number; speed: number;
    originalX: number; originalY: number;
  }
  const particlesRef = useRef<Particle[]>([])
  const mouseRef = useRef<{ x: number; y: number }>({ x: 0, y: 0 })

  // Phoenix shape coordinates generator
  const generatePhoenixShape = (centerX: number, centerY: number, count: number) => {
    const points: { x: number; y: number }[] = []
    const scale = 120
    
    // Head/beak area
    for (let i = 0; i < count * 0.1; i++) {
      const t = i / (count * 0.1)
      points.push({
        x: centerX + Math.sin(t * Math.PI) * 8,
        y: centerY - 120 - t * 20
      })
    }
    
    // Body (central vertical structure)
    for (let i = 0; i < count * 0.25; i++) {
      const t = i / (count * 0.25)
      points.push({
        x: centerX + Math.sin(t * Math.PI * 3) * 12,
        y: centerY - 100 + t * 140
      })
    }
    
    // Left wing
    for (let i = 0; i < count * 0.25; i++) {
      const t = i / (count * 0.25)
      const angle = t * Math.PI * 1.5
      const radius = 60 + Math.sin(angle * 2) * 25
      points.push({
        x: centerX - Math.cos(angle + Math.PI * 0.2) * radius - 40,
        y: centerY - Math.sin(angle + Math.PI * 0.2) * radius * 0.7 - 30
      })
    }
    
    // Right wing
    for (let i = 0; i < count * 0.25; i++) {
      const t = i / (count * 0.25)
      const angle = t * Math.PI * 1.5
      const radius = 60 + Math.sin(angle * 2) * 25
      points.push({
        x: centerX + Math.cos(angle + Math.PI * 0.2) * radius + 40,
        y: centerY - Math.sin(angle + Math.PI * 0.2) * radius * 0.7 - 30
      })
    }
    
    // Tail feathers (flowing downward and outward)
    for (let i = 0; i < count * 0.15; i++) {
      const t = i / (count * 0.15)
      const spread = Math.sin(t * Math.PI * 2) * 30
      points.push({
        x: centerX + spread + Math.sin(t * Math.PI * 6) * 15,
        y: centerY + 60 + t * 80
      })
    }
    
    // Fill remaining particles around the phoenix
    while (points.length < count) {
      const angle = Math.random() * Math.PI * 2
      const radius = 80 + Math.random() * 60
      points.push({
        x: centerX + Math.cos(angle) * radius,
        y: centerY + Math.sin(angle) * radius * 0.6
      })
    }
    
    return points
  }

  useEffect(() => {
    const handleWheel = (e: WheelEvent) => {
      if (!sectionRef.current) return
      
      const rect = sectionRef.current.getBoundingClientRect()
      
      // Only intercept when hero section is in view
      if (rect.top <= 0 && rect.bottom > 0) {
        e.preventDefault()
        
        // Accumulate scroll delta for phoenix progress
        accumulatedScroll.current += Math.abs(e.deltaY)
        
        // Calculate progress based on accumulated scroll (about 3 viewport heights to complete)
        const maxScroll = window.innerHeight * 3
        const progress = Math.max(0, Math.min(1, accumulatedScroll.current / maxScroll))
        
        setScrollProgress(progress)
        const complete = progress >= 0.95
        setPhoenixComplete(complete)
        
        if (complete && !allowScroll) {
          setAllowScroll(true)
        }
      }
    }

    const handleScroll = (e: Event) => {
      if (!sectionRef.current) return
      
      const rect = sectionRef.current.getBoundingClientRect()
      const currentScrollY = window.scrollY
      
      // If hero section is in view and phoenix isn't complete, lock scroll position
      if (rect.top <= 0 && rect.bottom > 0 && !phoenixComplete) {
        // Reset scroll to last known position
        if (currentScrollY !== lastScrollY.current && Math.abs(currentScrollY - lastScrollY.current) > 1) {
          window.scrollTo(0, lastScrollY.current)
        }
      } else if (phoenixComplete || rect.bottom <= 0) {
        // Update last scroll position when phoenix is complete or hero section is out of view
        lastScrollY.current = currentScrollY
      } else if (rect.top > 0) {
        // Hero section hasn't been reached yet, allow normal scrolling
        lastScrollY.current = currentScrollY
      }
    }

    const handleTouchMove = (e: TouchEvent) => {
      if (!sectionRef.current) return
      
      const rect = sectionRef.current.getBoundingClientRect()
      
      // Prevent touch scrolling when hero section is in view and phoenix isn't complete
      if (rect.top <= 0 && rect.bottom > 0 && !phoenixComplete) {
        e.preventDefault()
        
        // Simulate wheel event for touch scrolling
        const touch = e.touches[0]
        if (touch) {
          accumulatedScroll.current += 20 // Add fixed amount per touch move
          
          const maxScroll = window.innerHeight * 3
          const progress = Math.max(0, Math.min(1, accumulatedScroll.current / maxScroll))
          
          setScrollProgress(progress)
          const complete = progress >= 0.95
          setPhoenixComplete(complete)
          
          if (complete && !allowScroll) {
            setAllowScroll(true)
          }
        }
      }
    }

    // Initialize last scroll position
    lastScrollY.current = window.scrollY

    // Use passive: false to allow preventDefault
    document.addEventListener('wheel', handleWheel, { passive: false })
    document.addEventListener('scroll', handleScroll, { passive: false })
    document.addEventListener('touchmove', handleTouchMove, { passive: false })
    
    return () => {
      document.removeEventListener('wheel', handleWheel)
      document.removeEventListener('scroll', handleScroll)
      document.removeEventListener('touchmove', handleTouchMove)
    }
  }, [allowScroll, phoenixComplete])

  useEffect(() => {
    const canvas = canvasRef.current
    if (!canvas) return

    const ctx = canvas.getContext('2d')
    if (!ctx) return
    const particles: Particle[] = []
    const COUNT = 150

    // Set canvas size
    const resizeCanvas = () => {
      canvas.width = window.innerWidth
      canvas.height = window.innerHeight
    }
    resizeCanvas()
    window.addEventListener('resize', resizeCanvas)

    // Generate phoenix shape coordinates
    const phoenixPoints = generatePhoenixShape(canvas.width / 2, canvas.height / 2, COUNT)

    // Initialize particles
    for (let i = 0; i < COUNT; i++) {
      const originalX = Math.random() * canvas.width
      const originalY = Math.random() * canvas.height
      const phoenixPoint = phoenixPoints[i] || phoenixPoints[phoenixPoints.length - 1]
      
      particles.push({
        x: originalX,
        y: originalY,
        targetX: originalX,
        targetY: originalY,
        originalX,
        originalY,
        phoenixX: phoenixPoint.x + (Math.random() - 0.5) * 10, // Add slight randomness
        phoenixY: phoenixPoint.y + (Math.random() - 0.5) * 10,
        size: Math.random() * 2 + 1,
        opacity: Math.random() * 0.8 + 0.2,
        speed: Math.random() * 0.03 + 0.02
      })
    }
    particlesRef.current = particles

    // Animation loop
    const animate = (time: number) => {
      ctx.clearRect(0, 0, canvas.width, canvas.height)
      
      // Create dynamic gradient based on scroll progress
      const gradient = ctx.createRadialGradient(
        canvas.width / 2, canvas.height / 2, 0,
        canvas.width / 2, canvas.height / 2, canvas.width / 2
      )
      
      if (scrollProgress > 0) {
        const orangeIntensity = scrollProgress * 0.4
        gradient.addColorStop(0, `rgba(249, 115, 22, ${orangeIntensity})`)
        gradient.addColorStop(0.3, `rgba(194, 65, 12, ${orangeIntensity * 0.8})`)
        gradient.addColorStop(0.7, `rgba(59, 130, 246, ${0.1 * (1 - scrollProgress)})`)
        gradient.addColorStop(1, `rgba(0, 0, 0, ${0.4 + scrollProgress * 0.3})`)
      } else {
        gradient.addColorStop(0, 'rgba(59, 130, 246, 0.1)')
        gradient.addColorStop(1, 'rgba(0, 0, 0, 0.3)')
      }
      
      ctx.fillStyle = gradient
      ctx.fillRect(0, 0, canvas.width, canvas.height)

      particles.forEach((particle, i) => {
        // Determine target based on scroll progress and hover state
        let targetX, targetY
        
        if (scrollProgress > 0) {
          // Interpolate between original position and phoenix position based on scroll
          const easeProgress = scrollProgress * scrollProgress * (3 - 2 * scrollProgress) // Smooth ease
          targetX = particle.originalX + (particle.phoenixX - particle.originalX) * easeProgress
          targetY = particle.originalY + (particle.phoenixY - particle.originalY) * easeProgress
        } else if (hovered) {
          targetX = particle.phoenixX
          targetY = particle.phoenixY
        } else {
          targetX = particle.targetX
          targetY = particle.targetY
        }

        // Smooth movement towards target with increased speed during transformation
        const movementSpeed = particle.speed * (1 + scrollProgress * 2)
        particle.x += (targetX - particle.x) * movementSpeed
        particle.y += (targetY - particle.y) * movementSpeed

        // Add floating motion (reduced during phoenix formation)
        const floatIntensity = Math.max(0.1, 1 - scrollProgress * 0.8)
        particle.x += Math.sin(time * 0.001 + i * 0.1) * 0.5 * floatIntensity
        particle.y += Math.cos(time * 0.0005 + i * 0.1) * 0.3 * floatIntensity

        // Mouse interaction (reduced during phoenix formation)
        if (scrollProgress < 0.3) {
          const dx = mouseRef.current.x - particle.x
          const dy = mouseRef.current.y - particle.y
          const distance = Math.sqrt(dx * dx + dy * dy)
          if (distance < 100) {
            const force = (100 - distance) / 100 * (1 - scrollProgress * 2)
            particle.x -= dx * force * 0.1
            particle.y -= dy * force * 0.1
          }
        }

        // Enhanced color interpolation
        const progress = Math.min(1, scrollProgress * 1.2) // Slightly faster color change
        const blueColor = { r: 59, g: 130, b: 246 }
        const orangeColor = { r: 249, g: 115, b: 22 }
        
        const r = Math.round(blueColor.r + (orangeColor.r - blueColor.r) * progress)
        const g = Math.round(blueColor.g + (orangeColor.g - blueColor.g) * progress)
        const b = Math.round(blueColor.b + (orangeColor.b - blueColor.b) * progress)

        // Draw particle with enhanced glow effect
        ctx.save()
        ctx.globalAlpha = Math.min(1, particle.opacity + scrollProgress * 0.4)
        const color = `rgb(${r}, ${g}, ${b})`
        ctx.fillStyle = color
        ctx.shadowColor = color
        ctx.shadowBlur = 8 + scrollProgress * 25
        
        // Add inner glow for phoenix particles
        if (scrollProgress > 0.5) {
          ctx.shadowBlur = 30
          ctx.beginPath()
          ctx.arc(particle.x, particle.y, particle.size + scrollProgress * 2, 0, Math.PI * 2)
          ctx.fill()
        }
        
        ctx.beginPath()
        ctx.arc(particle.x, particle.y, particle.size + scrollProgress * 1.5, 0, Math.PI * 2)
        ctx.fill()
        ctx.restore()

        // Draw connections for phoenix formation
        if (scrollProgress > 0.2) {
          particles.forEach((otherParticle, j) => {
            if (i >= j) return
            
            const dist = Math.sqrt((particle.x - otherParticle.x) ** 2 + (particle.y - otherParticle.y) ** 2)
            const maxConnectionDist = 60 + scrollProgress * 40
            
            if (dist < maxConnectionDist) {
              ctx.save()
              ctx.globalAlpha = (scrollProgress - 0.2) * 0.6 * (1 - dist / maxConnectionDist)
              ctx.strokeStyle = `rgb(${r}, ${g}, ${b})`
              ctx.lineWidth = Math.max(0.5, scrollProgress * 2)
              ctx.beginPath()
              ctx.moveTo(particle.x, particle.y)
              ctx.lineTo(otherParticle.x, otherParticle.y)
              ctx.stroke()
              ctx.restore()
            }
          })
        }
      })

      // Add phoenix completion flash effect
      if (phoenixComplete && scrollProgress > 0.95) {
        const flashIntensity = Math.sin(time * 0.01) * 0.1 + 0.1
        ctx.save()
        ctx.globalAlpha = flashIntensity
        ctx.fillStyle = 'rgba(249, 115, 22, 0.3)'
        ctx.fillRect(0, 0, canvas.width, canvas.height)
        ctx.restore()
      }

      animationRef.current = requestAnimationFrame(animate)
    }

    animate(0)

    // Mouse tracking
    const handleMouseMove = (e: MouseEvent) => {
      mouseRef.current.x = e.clientX
      mouseRef.current.y = e.clientY
    }
    window.addEventListener('mousemove', handleMouseMove)

    return () => {
      window.removeEventListener('resize', resizeCanvas)
      window.removeEventListener('mousemove', handleMouseMove)
      if (animationRef.current !== null) {
        cancelAnimationFrame(animationRef.current)
      }
    }
  }, [hovered, scrollProgress, phoenixComplete])

  return (
    <section 
      ref={sectionRef}
      className="relative w-full h-screen overflow-hidden bg-gradient-to-br from-gray-900 via-blue-900 to-black"
    >
      {/* Animated background canvas */}
      <canvas
        ref={canvasRef}
        className="absolute inset-0 w-full h-full"
        onMouseEnter={() => setHovered(true)}
        onMouseLeave={() => setHovered(false)}
      />
      
      {/* Dynamic overlay based on scroll progress */}
      <div 
        className="absolute inset-0 transition-all duration-1000"
        style={{
          background: scrollProgress > 0 
            ? `radial-gradient(circle at center, rgba(249, 115, 22, ${scrollProgress * 0.2}) 0%, rgba(0,0,0,${0.3 + scrollProgress * 0.3}) 70%)`
            : 'linear-gradient(to bottom, rgba(0,0,0,0.4) 0%, transparent 50%, rgba(0,0,0,0.6) 100%)'
        }}
      />
      
      {/* Content */}
      <div className="relative z-10 flex flex-col items-center justify-center h-full text-center px-4">
        <div 
          className={`transform transition-all duration-700 ${
            scrollProgress > 0.7 ? 'opacity-20 scale-90' : scrollProgress > 0.3 ? 'opacity-60 scale-95' : 'hover:scale-105'
          }`}
        >
          <h1 className="text-5xl md:text-7xl font-bold text-white mb-6 leading-tight">
            <span 
              className="bg-gradient-to-r bg-clip-text text-transparent transition-all duration-1000"
              style={{
                backgroundImage: scrollProgress > 0.1
                  ? `linear-gradient(135deg, 
                      rgb(${59 + scrollProgress * 190}, ${130 - scrollProgress * 15}, ${246 - scrollProgress * 224}), 
                      rgb(${147 + scrollProgress * 102}, ${51 + scrollProgress * 64}, ${234 - scrollProgress * 212}), 
                      rgb(249, 115, 22))`
                  : 'linear-gradient(135deg, #3b82f6, #9333ea, #f97316)'
              }}
            >
              Igniting Impact,
            </span>
            <br />
            <span className="text-white drop-shadow-2xl">
              One Line of Code at a Time
            </span>
          </h1>
          
          <p className="text-xl md:text-2xl text-gray-300 mb-12 max-w-3xl leading-relaxed">
            Student-powered digital solutions for nonprofits.
            <br />
            <span className="text-lg text-gray-400 mt-2 block">
              Where innovation meets purpose, and code creates change.
            </span>
          </p>
          
          <div className="flex flex-col sm:flex-row gap-4 justify-center items-center">
            <button className="group px-8 py-4 bg-gradient-to-r from-blue-600 to-purple-600 hover:from-blue-700 hover:to-purple-700 text-white rounded-xl transition-all duration-300 font-semibold text-lg shadow-lg hover:shadow-xl hover:scale-105 transform">
              <span className="flex items-center gap-2">
                View Our Work
                <svg className="w-5 h-5 group-hover:translate-x-1 transition-transform" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M13 7l5 5m0 0l-5 5m5-5H6" />
                </svg>
              </span>
            </button>
            
            <button className="group px-8 py-4 border-2 border-white/30 hover:border-white hover:bg-white/10 rounded-xl transition-all duration-300 text-white font-semibold text-lg backdrop-blur-sm hover:scale-105 transform">
              <span className="flex items-center gap-2">
                Get Started
                <svg className="w-5 h-5 group-hover:rotate-45 transition-transform" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                  <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 4v16m8-8H4" />
                </svg>
              </span>
            </button>
          </div>
        </div>
        
        {/* Enhanced scroll indicator with phoenix status */}
        <div 
          className={`absolute bottom-8 left-1/2 transform -translate-x-1/2 transition-all duration-500 ${
            phoenixComplete ? 'animate-pulse scale-110' : 'animate-bounce'
          }`}
          style={{
            color: scrollProgress > 0.3 ? '#f97316' : 'rgba(255,255,255,0.6)'
          }}
        >
          <div className="flex flex-col items-center gap-2">
            <div className="text-center">
              <div className="text-sm font-medium mb-1">
                {phoenixComplete 
                  ? '🔥 Phoenix Risen - Continue Your Journey' 
                  : `🐣 Awakening Phoenix ${Math.round(scrollProgress * 100)}%`
                }
              </div>
              {!phoenixComplete && (
                <div className="w-32 bg-white/20 rounded-full h-1 mb-2">
                  <div 
                    className="bg-gradient-to-r from-blue-500 to-orange-500 h-1 rounded-full transition-all duration-300"
                    style={{ width: `${scrollProgress * 100}%` }}
                  />
                </div>
              )}
            </div>
            <svg 
              className={`w-6 h-6 ${phoenixComplete ? 'animate-bounce' : ''}`} 
              fill="none" 
              stroke="currentColor" 
              viewBox="0 0 24 24"
            >
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 14l-7 7m0 0l-7-7m7 7V3" />
            </svg>
          </div>
        </div>
      </div>
      
      {/* Responsive decorative phoenix embers */}
      {[...Array(6)].map((_, i) => (
        <div
          key={i}
          className="absolute rounded-full blur-sm animate-pulse transition-all duration-1000"
          style={{
            left: `${10 + (i * 15) % 80}%`,
            top: `${15 + (i * 25) % 70}%`,
            width: `${8 + (i % 3) * 8 + scrollProgress * 12}px`,
            height: `${8 + (i % 3) * 8 + scrollProgress * 12}px`,
            backgroundColor: scrollProgress > 0.2
              ? `rgba(249, 115, 22, ${0.3 + scrollProgress * 0.2 + (i % 2) * 0.1})`
              : `rgba(59, 130, 246, ${0.2 + (i % 2) * 0.1})`,
            animationDelay: `${i * 200}ms`,
            transform: `scale(${1 + scrollProgress * 0.5})`
          }}
        />
      ))}
    </section>
  )
}

export default HeroSection
